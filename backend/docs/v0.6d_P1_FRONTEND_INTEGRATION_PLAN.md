## 一、目标

在现有 `/stamp-hybrid-design` 页面新增一个 **"One-Click STAMP Demo"** 面板。

点击按钮 → 调用 `GET /api/v1/stamp/demo-one` → 展示完整 STAMP 分子信息，严格遵守科学边界。

### 1.1 硬约束 (不可违反)

| # | 约束 |
|:--|:----|
| R1 | **不修改** v0.6c 前端现有页面结构（仅追加新面板） |
| R2 | **不引入** 新的 npm 依赖 |
| R3 | **不显示** ipTM / pDockQ / docking_score / ΔG / MIC / MBC 数值 |

### 3.1 新增类型 `src/types/stamp.ts`

```typescript
// ───────────────────────────────────────────────
// v0.6d — Live backend StampCandidate (from /api/v1/stamp/demo-one)
// ───────────────────────────────────────────────

export interface StampDemoTargetingDomain {
  name: string;          // e.g. "OPRF_0001"
  sequence: string;      // e.g. "DKTKKAFLIAAG"
  length: number;
  net_charge: number;
  gravy: number | null;
}

export interface StampDemoLinker {
  name: string;          // e.g. "EAAAK"
  sequence: string;
  length: number;        // always 5 for EAAAK
  type: string;          // "rigid"
}

export interface StampDemoKillingDomain {
  name: string;          // e.g. "P4"
  sequence: string;      // e.g. "FSRFLRRVRRYRPKISFNLEPFFKF"
  length: number;
  net_charge: number | null;
  gravy: number | null;
}

export interface StampDemoBiophysical {
  length: number;
  net_charge: number | null;
  pI: number | null;
  GRAVY: number | null;
  hydrophobicity_fraction: number | null;
}

// v0.6d backend returns null — frontend must handle optional
export interface StampDemoExperimental {
  MIC_ug_ml: number | null;
  MBC_ug_ml: number | null;
  hemolysis_percent: number | null;
  LPS_binding_Kd_nM: number | null;
  pLDDT: number | null;
  ipTM: number | null;
  pDockQ: number | null;
  note: string | null;
}

export interface StampDemoCandidate {
  candidate_id: string;
  target_molecule: string;
  targeting_domain: StampDemoTargetingDomain;
  linker: StampDemoLinker;
  killing_domain: StampDemoKillingDomain;
  orientation: string;              // "N-to-C"
  terminal_modification: string;    // "-NH2"
  raw_full_sequence: string;        // "DKTKKAFLIAAGEAAAKFSRFLRRVRRYRPKISFNLEPFFKF"
  display_full_sequence: string;    // "DKTKKAFLIAAG-EAAAK-FSRFLRRVRRYRPKISFNLEPFFKF-NH2"
  is_complete: boolean;
  biophysical: StampDemoBiophysical;
  mock_scores: null;                // must be null in v0.6d
  experimental: StampDemoExperimental;
  structure_status: {
    monomer_predicted: boolean;
    complex_predicted: boolean;
    experimental_structure: boolean;
  };
  validation_status: string;        // "NOT_EXPERIMENTALLY_VALIDATED"
}

// Backend ApiResponse envelope: { code, message, data }
export interface StampDemoApiResponse {
  code: number;
  message: string;
  data: StampDemoCandidate;
}
```

> **重要**: 不要在 `StampHybridCandidate` 上修改字段类型。v0.6b 的前端 mock 类型与 v0.6d 后端 DTO 是两套不同的 schema。用独立类型避免污染。

## 四、API 客户端 (F1)

### 4.1 `src/lib/stampApi.ts`

```typescript
/**
 * STAMP Platform v0.6d — Backend API client
 * Raw fetch wrappers for the FastAPI backend.
 *
 * Usage:
 *   import { fetchStampDemoOne } from '@/lib/stampApi';
 *   const result = await fetchStampDemoOne();
 */

import type { StampDemoApiResponse, StampDemoCandidate } from '@/types/stamp';

const BACKEND_BASE_URL = '/api/v1'; // 开发环境 Vite proxy, 生产环境 nginx

/**
 * Call GET /api/v1/stamp/demo-one.
 * Auto-selects the top-ranked PepMLM candidate (lowest PPL) + P4 + EAAAK.
 *
 * @returns Parsed StampDemoCandidate on success.
 * @throws {Error} on network error or non-200 response.
 */
export async function fetchStampDemoOne(): Promise<StampDemoCandidate> {
  const url = `${BACKEND_BASE_URL}/stamp/demo-one`;

  const response = await fetch(url, {
    method: 'GET',
    headers: { 'Accept': 'application/json' },
  });

  if (!response.ok) {
    const text = await response.text();
    throw new Error(
      `Backend returned ${response.status}: ${text.slice(0, 200)}`
    );
  }

  const json: StampDemoApiResponse = await response.json();

  if (json.code !== 200) {
    throw new Error(
      `Backend error code ${json.code}: ${json.message ?? 'Unknown'}`
    );
  }

  // v0.6d guard: mock_scores must be null
  if (json.data.mock_scores !== null) {
    console.warn(
      '[stampApi] mock_scores is not null — this violates v0.6d scientific integrity rules',
      json.data.mock_scores
    );
  }

  return json.data;
}
```

### 5.1 `src/components/stamp/OneClickStampDemo.tsx`

```tsx
import { useState, useCallback } from 'react';
import { useLanguage } from '@/i18n/LanguageContext';
import { fetchStampDemoOne } from '@/lib/stampApi';
import type { StampDemoCandidate } from '@/types/stamp';

/**
 * One-Click STAMP Demo — calls GET /api/v1/stamp/demo-one
 * and displays the assembled STAMP molecule with full scientific
 * integrity guardrails.
 */
export default function OneClickStampDemo() {
  const { t } = useLanguage();
  const s = (t.stampHybridDesign as Record<string, unknown>)
    ?.oneClickDemo as Record<string, string> | undefined;

  const [candidate, setCandidate] = useState<StampDemoCandidate | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const handleRun = useCallback(async () => {
    setLoading(true);
    setError(null);
    setCandidate(null);
    try {
      const result = await fetchStampDemoOne();
      setCandidate(result);
    } catch (err) {
      const msg = err instanceof Error ? err.message : 'Unknown error';
      setError(msg);
    } finally {
      setLoading(false);
    }
  }, []);

  return (
    <div className="mb-6 rounded-lg border bg-white shadow-sm">
      {/* Header */}
      <div className="flex items-center justify-between border-b px-4 py-3">
        <div>
          <h3 className="font-semibold text-slate-800">
            {s?.title ?? 'One-Click STAMP Demo'}
          </h3>
          <p className="text-xs text-slate-500">
            {s?.subtitle ?? 'Live backend assembly via /api/v1/stamp/demo-one'}
          </p>
        </div>
        <button
          onClick={handleRun}
          disabled={loading}
          className={`
            rounded-lg px-4 py-2 text-sm font-medium text-white
            transition-colors
            ${loading
              ? 'cursor-not-allowed bg-slate-400'
              : 'bg-xh-primary hover:bg-xh-primary/90'
            }
          `}
        >
          {loading
            ? (s?.buttonLoading ?? 'Assembling...')
            : (s?.buttonRun ?? 'Run One Candidate End-to-End')}
        </button>
      </div>

      {/* Loading */}
      {loading && (
        <div className="flex items-center justify-center py-8">
          <div className="h-6 w-6 animate-spin rounded-full border-2 border-xh-primary border-t-transparent" />
          <span className="ml-3 text-sm text-slate-500">
            {s?.loadingMsg ?? 'Assembling STAMP candidate from backend...'}
          </span>
        </div>
      )}

      {/* Error */}
      {error && (
        <div className="m-4 rounded-lg border border-red-200 bg-red-50 p-3">
          <div className="text-sm font-medium text-red-700">
            {s?.errorTitle ?? 'Backend unavailable'}
          </div>
          <div className="mt-1 text-xs text-red-500 font-mono">{error}</div>
          <div className="mt-2 text-xs text-red-400">
            {s?.errorHint ?? 'Ensure the backend is running: cd backend && uvicorn app.main:app --host 0.0.0.0 --port 8000'}
          </div>
        </div>
      )}

      {/* Result */}
      {candidate && !loading && (
        <div className="p-4 space-y-4">
          {/* Three-domain visualization */}
          <div className="flex items-center gap-2">
            {/* Targeting */}
            <div className="flex-1 rounded-lg bg-teal-50 border border-teal-200 p-3">
              <div className="text-xs text-teal-700 font-medium">
                {s?.targetingLabel ?? 'Targeting Peptide'}
              </div>
              <div className="mt-1 text-xs font-mono text-teal-900 truncate" title={candidate.targeting_domain.sequence}>
                {candidate.targeting_domain.name}: {candidate.targeting_domain.sequence}
              </div>
            </div>

            {/* Arrow + Linker */}
            <div className="flex items-center gap-1 text-slate-400">
              <span>→</span>
              <div className="rounded bg-amber-50 border border-amber-200 px-2 py-1">
                <span className="text-xs font-mono text-amber-700">
                  {candidate.linker.sequence}
                </span>
              </div>
              <span>→</span>
            </div>

            {/* Killing */}
            <div className="flex-1 rounded-lg bg-red-50 border border-red-200 p-3">
              <div className="text-xs text-red-700 font-medium">
                {s?.ampLabel ?? 'AMP (Killing Domain)'}
              </div>
              <div className="mt-1 text-xs font-mono text-red-900 truncate" title={candidate.killing_domain.sequence}>
                {candidate.killing_domain.name}: {candidate.killing_domain.sequence}
              </div>
            </div>
          </div>

          {/* Full STAMP sequence */}
          <div className="rounded-lg bg-slate-50 border p-3">
            <div className="text-xs text-slate-500">
              {s?.fullSequenceLabel ?? 'Full STAMP Sequence (with C-terminal amidation)'}
            </div>
            <div className="mt-1 text-sm font-mono text-slate-800 break-all">
              {candidate.display_full_sequence}
            </div>
            <div className="mt-1 text-xs text-slate-400 font-mono">
              Raw: {candidate.raw_full_sequence}
            </div>
          </div>

          {/* Biophysical properties grid */}
          <div className="grid grid-cols-4 gap-3">
            <MetricItem
              label={s?.lengthLabel ?? 'Length'}
              value={candidate.biophysical.length}
            />
            <MetricItem
              label={s?.netChargeLabel ?? 'Net Charge'}
              value={candidate.biophysical.net_charge}
            />
            <MetricItem
              label={s?.piLabel ?? 'pI'}
              value={candidate.biophysical.pI}
              fallback={s?.notComputed ?? '—'}
            />
            <MetricItem
              label={s?.gravyLabel ?? 'GRAVY'}
              value={candidate.biophysical.GRAVY}
              fallback={s?.notComputed ?? '—'}
            />
          </div>

          {/* Validation status + scientific boundary */}
          <div className="rounded-lg border border-amber-200 bg-amber-50 p-3">
            <div className="flex items-start gap-2">
              <span className="text-amber-600 text-sm mt-0.5">⚠️</span>
              <div>
                <div className="text-sm font-medium text-amber-800">
                  {s?.validationLabel ?? 'Validation Status'}:{' '}
                  <code className="rounded bg-amber-100 px-1.5 py-0.5 text-xs font-bold text-amber-900">
                    {candidate.validation_status}
                  </code>
                </div>
                <div className="mt-1 text-xs text-amber-700">
                  {s?.boundaryWarning ??
                    'This candidate is generated by AI and assembled computationally. No wet-lab experiments have been performed. All structural (ipTM, pDockQ) and antimicrobial (MIC, MBC) metrics are pending experimental validation.'}
                </div>
                {candidate.mock_scores !== null && (
                  <div className="mt-2 text-xs text-red-600 font-bold">
                    ⚠️ mock_scores non-null detected — potential scientific integrity violation
                  </div>
                )}
              </div>
            </div>
          </div>

          {/* Negative space: explicitly show what is NOT available */}
          <div className="text-xs text-slate-400 space-y-1">
            <div>
              {s?.noDataPrefix ?? 'Not available'}:
              ipTM, pDockQ, binding ΔG, docking score, MIC, MBC, hemolysis %
            </div>
            <div>
              {s?.noDataReason ??
                'All experimental data fields are intentionally null. Real measurements require wet-lab assays.'}
            </div>
          </div>
        </div>
      )}
    </div>
  );
}

/** Small helper for displaying a metric value (or fallback). */
function MetricItem({
  label,
  value,
  fallback = '—',
}: {
  label: string;
  value: number | null | undefined;
  fallback?: string;
}) {
  const display =
    value !== null && value !== undefined
      ? typeof value === 'number'
        ? value % 1 === 0
          ? value.toString()
          : value.toFixed(2)
        : String(value)
      : fallback;

  return (
    <div className="rounded-lg bg-white border p-3 text-center">
      <div className="text-2xl font-bold text-slate-800">{display}</div>
      <div className="text-xs text-slate-500">{label}</div>
    </div>
  );
}
```

## 九、风险与规避

| 风险 | 影响 | 规避 |
|:----|:----|:----|
| 后端 proto 字段名与前端类型不匹配 | 运行时 undefined | 使用 Optional chaining + 测试 mock 数据 |
| `mock_scores` 在 Pydantic `exclude_none` 策略下被省去 | 前端 `candidate.mock_scores` 为 undefined 而非 null | 前端同时检查 `null` 和 `undefined` |
| Vite proxy 未正确配置 | fetch 404 | 错误提示中包含 proxy 配置说明 |
| `biophysical.pI` / `biophysical.GRAVY` 为 None | 显示空值 | `MetricItem` fallback 为 "—" 或 "未计算" |
| 后端 CORS 未开放 | 502/跨域错误 | FastAPI main.py 中已配置 `allow_origins=["*"]`（确认不会被注释） |
